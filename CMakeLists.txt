cmake_minimum_required(VERSION 3.20)
project(GTASANetworkLauncher LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

option(GTANET_BUILD_SERVER "Build dedicated server" ON)
option(GTANET_BUILD_CLIENT "Build GTA SA Windows client DLL" ${WIN32})
option(GTANET_BUILD_LAUNCHER "Build Windows launcher" ${WIN32})
option(GTANET_BUILD_TESTS "Build protocol/server tests" OFF)

if(GTANET_BUILD_CLIENT)
  if(NOT WIN32)
    message(FATAL_ERROR "The GTA client DLL is Windows-only")
  endif()
  add_subdirectory(client_dll)
endif()
if(GTANET_BUILD_LAUNCHER)
  if(NOT WIN32)
    message(FATAL_ERROR "The launcher is Windows-only")
  endif()
  add_subdirectory(launcher)
endif()
if(GTANET_BUILD_SERVER)
  add_subdirectory(server)
endif()
if(GTANET_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
