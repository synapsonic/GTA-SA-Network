add_executable(gtasa_server main.cpp ServerConfig.cpp AccountStore.cpp LuaResource.cpp ResourceRuntime.cpp ServerRuntime.cpp SqliteStore.cpp)
target_include_directories(gtasa_server PRIVATE ../shared)
if(WIN32)
  target_link_libraries(gtasa_server PRIVATE ws2_32)
endif()
option(GTANET_WITH_LUA "Enable Lua 5.4 server resources" OFF)
if(GTANET_WITH_LUA)
  find_package(Lua 5.4 REQUIRED)
  target_compile_definitions(gtasa_server PRIVATE GTANET_WITH_LUA=1)
  target_include_directories(gtasa_server PRIVATE ${LUA_INCLUDE_DIR})
  target_link_libraries(gtasa_server PRIVATE ${LUA_LIBRARIES})
endif()
option(GTANET_WITH_SQLITE "Enable SQLite persistence" OFF)
if(GTANET_WITH_SQLITE)
  find_package(SQLite3 REQUIRED)
  target_compile_definitions(gtasa_server PRIVATE GTANET_WITH_SQLITE=1)
  target_link_libraries(gtasa_server PRIVATE SQLite::SQLite3)
endif()
